    79→    79→/** Split narration into single-line chunks that never exceed maxChars. */
    80→    80→export function chunkSingleLine(text: string, maxChars: number): string[] {
    81→    81→  const words = text.replace(/\s+/g, " ").trim().split(" ").filter(Boolean);
    82→    82→  const chunks: string[] = [];
    83→    83→  let buffer: string[] = [];
    84→    84→  for (const w of words) {
    85→    85→    const next = buffer.length ? `${buffer.join(" ")} ${w}` : w;
    86→    86→    if (next.length > maxChars && buffer.length) {
    87→    87→      chunks.push(buffer.join(" "));
    88→    88→      buffer = [w];
    89→    89→    } else {
    90→    90→      buffer.push(w);
    91→    91→    }
    92→    92→  }
    93→    93→  if (buffer.length) chunks.push(buffer.join(" "));
    94→    94→  // Pecahan terakhir yang terlalu pendek digabung ke chunk sebelumnya agar
    95→    95→  // caption tidak menampilkan satu-dua kata sendirian. Batas gabungan harus
    96→    96→  // tetap muat SATU baris caption renderer (maks ~1120px: B1 34px ≈ 17.1px/
    97→    97→  // karakter → ≤58 karakter; B2 40px ≈ 20.1px/karakter → ≤50 karakter).
    98→    98→  if (chunks.length >= 2 && chunks[chunks.length - 1].length < 9) {
    99→    99→    const last = chunks.pop()!;
   100→   100→    const merged = `${chunks[chunks.length - 1]} ${last}`;
   101→   101→    const cap = Math.min(50, maxChars + 8);
   102→   102→    if (merged.length <= cap) chunks[chunks.length - 1] = merged;
   103→   103→    else chunks.push(last);
   104→   104→  }
   105→   105→  return chunks;
   106→   106→}

   108→   108→/**
   109→   109→ * Sequential non-overlapping single-line cues between startT and endT,
   110→   110→ * weighted by chunk length. Semua chunk SELALU tampil: bila jendela terlalu
   111→   111→ * pendek untuk durasi natural, semua durasi dikompres proporsional (bukan
   112→   112→ * dipotong) sehingga tidak ada caption yang hilang. Enforces the renderer's
   113→   113→ * caption rules.
   114→   114→ */
   115→   115→export function buildSingleLineCues(text: string, startT: number, endT: number, maxChars: number): CanvasCue[] {
   116→   116→  const chunks = chunkSingleLine(text, maxChars);
   117→   117→  if (!chunks.length) return [];
   118→   118→  const usable = Math.max(0.8, endT - startT - 0.25);
   119→   119→  const weights = chunks.map((c) => Math.max(4, c.length));
   120→   120→  const sum = weights.reduce((a, b) => a + b, 0);
   121→   121→  const naturals = chunks.map((_, i) => Math.max(0.7, (weights[i] / sum) * usable));
   122→   122→  const totalNatural = naturals.reduce((a, b) => a + b, 0);
   123→   123→  const scale = totalNatural > usable ? usable / totalNatural : 1;
   124→   124→  const cues: CanvasCue[] = [];
   125→   125→  for (let i = 0; i < chunks.length; i++) {
   126→   126→    const dur = Math.max(0.45, naturals[i] * scale);
   127→   127→    const start = i === 0 ? startT : cues[i - 1].end;
   128→   128→    let end = Math.min(start + dur, endT);
   129→   129→    if (end - start < 0.35) end = Math.min(endT, start + 0.35);
   130→   130→    if (end <= start) break;
   131→   131→    cues.push({ text: chunks[i], start: round3(start), end: round3(end) });
   132→   132→  }
   133→   133→  return cues;
   134→   134→}
   135→   135→
   136→   136→function round3(x: number): number {
   137→   137→  return Number(x.toFixed(3));
   138→   138→}
   139→   139→
   140→   140→/**
   141→   141→ * Renderer menuntut key times strictly increasing dan t di dalam span.
   142→   142→ * Normalisasi: buang key di luar span, gabungkan properti pada t yang sama,
   143→   143→ * lalu urutkan. Dipakai untuk SEMUA layer agar mustahil duplikat t.
   144→   144→ */
   145→   145→function normalizeKeys<T extends { t: number }>(keys: T[], span: number): T[] {
   146→   146→  const byT = new Map<number, T>();
   147→   147→  for (const k of keys) {
   148→   148→    if (k.t < 0 || k.t > span - 0.05) continue;
   149→   149→    const existing = byT.get(k.t);
   150→   150→    if (existing) Object.assign(existing, k);
   151→   151→    else byT.set(k.t, { ...k });
   152→   152→  }
   153→   153→  return [...byT.values()].sort((a, b) => a.t - b.t);
   154→   154→}

   374→   374→/** Split the scene span into beat windows weighted by label length. */
   375→   375→function planBeats(beats: AutopilotBeat[], span: number): BeatWin[] {
   376→   376→  const maxBeats = Math.max(1, Math.floor(span / 1.3));
   377→   377→  const list = beats.length <= maxBeats ? beats : beats.slice(0, maxBeats);
   378→   378→  const weights = list.map((b) => Math.max(6, b.label.length));
   379→   379→  const sum = weights.reduce((a, b) => a + b, 0);
   380→   380→  const wins: BeatWin[] = [];
   381→   381→  let acc = 0;
   382→   382→  for (let i = 0; i < list.length; i++) {
   383→   383→    const relStart = acc;
   384→   384→    acc += (weights[i] / sum) * span;
   385→   385→    wins.push({
   386→   386→      relStart: round3(relStart),
   387→   387→      relEnd: round3(i === list.length - 1 ? span : acc),
   388→   388→      beat: list[i],
   389→   389→      isLast: i === list.length - 1,
   390→   390→    });
   391→   391→  }
   392→   392→  return wins;
   393→   393→}

   395→   395→function fadeFor(relStart: number, relEnd: number, isLast: boolean): { inD: number; outD: number } {
   396→   396→  const win = relEnd - relStart;
   397→   397→  return {
   398→   398→    inD: Math.min(0.3, Math.max(0.12, win * 0.3)),
   399→   399→    outD: isLast ? 0 : Math.min(0.35, Math.max(0.14, win * 0.3)),
   400→   400→  };
   401→   401→}
   402→   402→
   403→   403→/**
   404→   404→ * Stickman per beat: crossfade masuk/keluar + bobbing & sway sepanjang
   405→   405→ * jendela beat (bukan cuma 6 detik pertama), sehingga gerakan terus hidup.
   406→   406→ */
   407→   407→function stickKeys(
   408→   408→  relStart: number,
   409→   409→  relEnd: number,
   410→   410→  baseY: number,
   411→   411→  baseX: number,
   412→   412→  fade: { inD: number; outD: number },
   413→   413→  span: number
   414→   414→): Keys {
   415→   415→  const keys: Keys = [];
   416→   416→  keys.push({ t: round3(relStart), alpha: 0, y: baseY + 12, x: baseX });
   417→   417→  keys.push({ t: round3(relStart + fade.inD), alpha: 1, y: baseY, x: baseX });
   418→   418→  const endT = Math.min(relEnd, span - 0.06);
   419→   419→  const period = 0.8;
   420→   420→  let t = relStart + fade.inD + period;
   421→   421→  let i = 0;
   422→   422→  while (t < endT - (fade.outD > 0 ? fade.outD : 0) - 0.08) {
   423→   423→    keys.push({
   424→   424→      t: round3(t),
   425→   425→      y: baseY + (i % 2 === 0 ? -18 : 0),
   426→   426→      x: baseX + (i % 2 === 1 ? 12 : -12),
   427→   427→    });
   428→   428→    t += period;
   429→   429→    i++;
   430→   430→  }
   431→   431→  if (fade.outD > 0) {
   432→   432→    keys.push({ t: round3(relEnd - fade.outD), alpha: 1 });
   433→   433→    keys.push({
   434→   434→      t: round3(Math.max(relEnd - 0.06, relEnd - fade.outD + 0.05)),
   435→   435→      alpha: 0,
   436→   436→      y: baseY + 12,
   437→   437→      x: baseX,
   438→   438→    });
   439→   439→  } else if (t - period < endT) {
   440→   440→    keys.push({ t: round3(t - period + 0.4), y: baseY + (i % 2 === 0 ? -18 : 0), x: baseX });
   441→   441→  }
   442→   442→  return normalizeKeys(keys, span);
   443→   443→}

   445→   445→/** Kartu kata kunci: meluncur naik saat beat mulai, pudar sebelum beat berikut. */
   446→   446→function cardKeys(relStart: number, relEnd: number, y: number, span: number, isLast: boolean): Keys {
   447→   447→  const keys: Keys = [];
   448→   448→  const win = relEnd - relStart;
   449→   449→  const inD = Math.min(0.3, Math.max(0.12, win * 0.4));
   450→   450→  keys.push({ t: round3(relStart + 0.1), alpha: 0, y: y + 26 });
   451→   451→  keys.push({ t: round3(relStart + 0.1 + inD + 0.35), alpha: 1, y });
   452→   452→  if (!isLast) {
   453→   453→    const outD = fadeFor(relStart, relEnd, false).outD;
   454→   454→    keys.push({ t: round3(relEnd - outD), alpha: 1 });
   455→   455→    keys.push({ t: round3(Math.max(relEnd - 0.06, relEnd - outD + 0.05)), alpha: 0 });
   456→   456→  }
   457→   457→  return normalizeKeys(keys, span);
   458→   458→}
   459→   459→
   460→   460→/** Garis/panah yang menggambar sendiri di awal beat, lalu pudar bersama kartu. */
   461→   461→function strokeKeys(relStart: number, relEnd: number, delay: number, span: number, isLast: boolean): Keys {
   462→   462→  const keys: Keys = [];
   463→   463→  keys.push({ t: round3(relStart + delay), progress: 0 });
   464→   464→  keys.push({ t: round3(Math.min(relStart + delay + 1.0, relEnd - 0.15)), progress: 1 });
   465→   465→  if (!isLast) {
   466→   466→    const outD = fadeFor(relStart, relEnd, false).outD;
   467→   467→    keys.push({ t: round3(relEnd - outD), alpha: 1 });
   468→   468→    keys.push({ t: round3(Math.max(relEnd - 0.06, relEnd - outD + 0.05)), alpha: 0 });
   469→   469→  }
   470→   470→  return normalizeKeys(keys, span);
   471→   471→}
   472→   472→
   473→   473→/** Prop krayon: melayang masuk dengan ayunan lembut, terikat pada jendela beat. */
   474→   474→function propKeys(relStart: number, relEnd: number, span: number, isLast: boolean): Keys {
   475→   475→  const keys: Keys = [];
   476→   476→  keys.push({ t: round3(relStart + 0.12), alpha: 0, angle: -0.07 });
   477→   477→  keys.push({ t: round3(relStart + 0.5), alpha: 1, angle: 0.03 });
   478→   478→  let t = relStart + 1.5;
   479→   479→  let i = 0;
   480→   480→  while (t < relEnd - 0.3) {
   481→   481→    keys.push({ t: round3(t), angle: i % 2 === 0 ? -0.04 : 0.03 });
   482→   482→    t += 1.1;
   483→   483→    i++;
   484→   484→  }
   485→   485→  if (!isLast) {
   486→   486→    const outD = fadeFor(relStart, relEnd, false).outD;
   487→   487→    keys.push({ t: round3(relEnd - outD), alpha: 1 });
   488→   488→    keys.push({ t: round3(Math.max(relEnd - 0.06, relEnd - outD + 0.05)), alpha: 0 });
   489→   489→  }
   490→   490→  return normalizeKeys(keys, span);
   491→   491→}
   492→   492→
   493→   493→/**
   494→   494→ * Build + write the full autopilot project (one shot per scene, 2-4 beats per
   495→   495→ * scene). Tiap beat punya kartu kata kunci + pose stickman + prop + aksen
   496→   496→ * yang muncul mengikuti urutan narasi — bukan visual statis satu scene.
   497→   497→ */
   498→   498→export function buildAutopilotProject(input: AutopilotCanvasInput): {
   499→   499→  projectPath: string;
   500→   500→  duration: number;
   501→   501→  cues: CanvasCue[];
   502→   502→} {
   503→   503→  const lead = 0.5;
   504→   504→  const tail = 0.9;
   505→   505→  const duration = input.voiceDuration + lead + tail;
   506→   506→  const plane = planeFor(input.ratio);
   507→   507→  const ink = input.styleB === "B2" ? INK_WARM : INK;
   508→   508→  const is916 = input.ratio === "9:16";
   509→   509→
   510→   510→  // Scene spans proportional to narration length. Semua batas langsung
   511→   511→  // dibulatkan ke 3 desimal agar identik dengan shot.start/end yang dibaca
   512→   512→  // validator renderer (span = round(end) - round(start)), sehingga key &
   513→   513→  // layer from/to tidak pernah melampaui span.
   514→   514→  const weights = input.scenes.map((s) => Math.max(10, s.narration.length));
   515→   515→  const weightSum = weights.reduce((a, b) => a + b, 0);
   516→   516→  const usable = Math.max(1, duration - lead - tail);
   517→   517→  const starts: number[] = [];
   518→   518→  let acc = 0;
   519→   519→  for (let i = 0; i < input.scenes.length; i++) {
   520→   520→    starts.push(round3(acc));
   521→   521→    acc += (weights[i] / weightSum) * usable;
   522→   522→  }
   523→   523→  const ends: number[] = input.scenes.map((_, i) =>
   524→   524→    i === input.scenes.length - 1 ? round3(duration) : starts[i + 1]
   525→   525→  );
   526→   526→
   527→   527→  const stickBaseX = is916 ? STICKMAN_X_916 : STICKMAN_X_169;
   528→   528→  const stickBaseY = plane.planeH * (is916 ? 0.36 : 0.3);
   529→   529→  const stickW = is916 ? 380 : 300;
   530→   530→  const propA = is916
   531→   531→    ? { x: plane.planeW - 430, y: 120, w: 320 }
   532→   532→    : { x: plane.planeW - 360, y: 64, w: 240 };
   533→   533→  const propB = is916
   534→   534→    ? { x: plane.planeW - 380, y: plane.planeH * 0.42, w: 270 }
   535→   535→    : { x: plane.planeW - 330, y: plane.planeH * 0.42, w: 220 };
   536→   536→
   537→   537→  const shots: Shot[] = [];
   538→   538→  const captions: CanvasCue[] = [];
   539→   539→  const sfx: NonNullable<CanvasProject["sfx"]> = [];
   540→   540→  const sfxKinds = ["pop", "chime", "paper", "tap"] as const;
   541→   541→
   542→   542→  input.scenes.forEach((scene, i) => {
   543→   543→    const shotStart = starts[i];
   544→   544→    const shotEnd = ends[i];
   545→   545→    const span = Math.max(0.8, shotEnd - shotStart);
   546→   546→    const layers: Layer[] = [];
   547→   547→
   548→   548→    // Title card on the opening scene.
   549→   549→    if (i === 0 && input.title && span >= 1.2) {
   550→   550→      layers.push({
   551→   551→        type: "text",
   552→   552→        text: input.title.slice(0, 80),
   553→   553→        x: 90,
   554→   554→        y: is916 ? plane.planeH * 0.1 : 60,
   555→   555→        w: is916 ? 720 : 780,
   556→   556→        size: is916 ? 84 : 76,
   557→   557→        bold: true,
   558→   558→        color: ink,
   559→   559→        keys: normalizeKeys(
   560→   560→          [
   561→   561→            { t: 0.2, alpha: 0, y: (is916 ? plane.planeH * 0.1 : 60) + 22 },
   562→   562→            { t: 0.8, alpha: 1, y: is916 ? plane.planeH * 0.1 : 60 },
   563→   563→            { t: Math.min(2.8, span - 0.7), alpha: 1 },
   564→   564→            { t: Math.min(3.4, span - 0.15), alpha: 0 },
   565→   565→          ],
   566→   566→          span
   567→   567→        ),
   568→   568→      });
   569→   569→    }
   570→   570→
   571→   571→    // Beats: fallback satu beat dari scene bila model tidak memberi beats.
   572→   572→    const beats: AutopilotBeat[] = scene.beats.length
   573→   573→      ? scene.beats
   574→   574→      : [
   575→   575→          {
   576→   576→            label: (scene.keyword || scene.narration.split(" ").slice(0, 3).join(" ")).slice(0, 34),
   577→   577→            pose: (Math.max(0, Math.min(2, Math.round(scene.pose))) as 0 | 1 | 2),
   578→   578→            asset: scene.asset,
   579→   579→            accent: scene.overlay,
   580→   580→          },
   581→   581→        ];
   582→   582→
   583→   583→    planBeats(beats, span).forEach((b, bi) => {
   584→   584→      const fade = fadeFor(b.relStart, b.relEnd, b.isLast);
   585→   585→      const beatWin = b.relEnd - b.relStart;
   586→   586→      const propAsset =
   587→   587→        b.beat.asset && b.beat.asset.startsWith("@b1/") ? b.beat.asset : null;
   588→   588→      const kwHasAsset = Boolean(propAsset);
   589→   589→
   590→   590→      // Stickman: pose berubah tiap beat, bobbing sepanjang beat. Atlas pose
   591→   591→      // hanya sah untuk gaya B1 (aturan renderer) — B2 memakai kartu + prop.
   592→   592→      if (input.styleB === "B1") {
   593→   593→        layers.push({
   594→   594→          type: "pose",
   595→   595→          pose: b.beat.pose,
   596→   596→          x: stickBaseX,
   597→   597→          y: stickBaseY,
   598→   598→          w: stickW,
   599→   599→          from: round3(b.relStart),
   600→   600→          // `to` memakai span mentah (identik dengan hitungan validator) dan
   601→   601→          // di-clamp — round3(span) bisa membulatkan ke ATAS dan melanggar.
   602→   602→          to: Math.min(b.isLast ? span : b.relEnd, span),
   603→   603→          keys: stickKeys(b.relStart, b.relEnd, stickBaseY, stickBaseX, fade, span),
   604→   604→        });
   605→   605→      }
   606→   606→
   607→   607→      // Kartu kata kunci per beat.
   608→   608→      const label = (b.beat.label || scene.keyword).slice(0, 34);
   609→   609→      if (label) {
   610→   610→        const kwY = plane.planeH * (i === 0 ? 0.3 : 0.27) + (bi % 2 === 1 ? (is916 ? 44 : 40) : 0);
   611→   611→        layers.push({
   612→   612→          type: "text",
   613→   613→          text: label,
   614→   614→          x: 560,
   615→   615→          y: kwY,
   616→   616→          w: kwHasAsset ? 340 : is916 ? 620 : 560,
   617→   617→          size: kwHasAsset ? 62 : is916 ? 72 : 64,
   618→   618→          bold: true,
   619→   619→          color: ink,
   620→   620→          keys: cardKeys(b.relStart, b.relEnd, kwY, span, b.isLast),
   621→   621→        });
   622→   622→        // Aksen menggambar sendiri: underline (umum) atau arrow (sorotan).
   623→   623→        const accent = b.beat.accent === "arrow" && beatWin >= 2.4 ? "arrow" : "underline";
   624→   624→        if (accent === "arrow") {
   625→   625→          layers.push({
   626→   626→            type: "arrow",
   627→   627→            x: 580,
   628→   628→            y: plane.planeH * (is916 ? 0.62 : 0.6),
   629→   629→            w: 320,
   630→   630→            color: ink,
   631→   631→            keys: strokeKeys(b.relStart, b.relEnd, 1.0, span, b.isLast),
   632→   632→          });
   633→   633→        } else if (beatWin >= 1.6) {
   634→   634→          layers.push({
   635→   635→            type: "underline",
   636→   636→            x: 560,
   637→   637→            y: kwY + (kwHasAsset ? 88 : 104),
   638→   638→            w: 300,
   639→   639→            lineWidth: 7,
   640→   640→            color: ink,
   641→   641→            keys: strokeKeys(b.relStart, b.relEnd, 0.9, span, b.isLast),
   642→   642→          });
   643→   643→        }
   644→   644→      }
   645→   645→
   646→   646→      // Prop krayon terikat beat: posisi bergantian kanan-atas / kanan-tengah.
   647→   647→      if (propAsset) {
   648→   648→        const pos = bi % 2 === 0 ? propA : propB;
   649→   649→        layers.push({
   650→   650→          type: "image",
   651→   651→          path: propAsset,
   652→   652→          x: pos.x,
   653→   653→          y: pos.y,
   654→   654→          w: pos.w,
   655→   655→          from: round3(b.relStart),
   656→   656→          to: Math.min(b.isLast ? span : b.relEnd, span),
   657→   657→          keys: propKeys(b.relStart, b.relEnd, span, b.isLast),
   658→   658→        });
   659→   659→      }
   660→   660→
   661→   661→      // SFX pop kecil di awal tiap beat (menegaskan ritme narasi).
   662→   662→      sfx.push({
   663→   663→        time: round2(shotStart + b.relStart + 0.12),
   664→   664→        kind: sfxKinds[(i * 2 + bi) % sfxKinds.length],
   665→   665→        gain: 0.12,
   666→   666→      });
   667→   667→    });
   668→   668→
   669→   669→    shots.push({ start: round3(shotStart), end: round3(shotEnd), layers });
   670→   670→
   671→   671→    // Scene caption: single-line chunks inside the scene window (semua chunk
   672→   672→    // selalu tampil — durasi dikompres bila jendela sempit).
   673→   673→    const maxChars = is916 ? (input.styleB === "B1" ? 44 : 36) : input.styleB === "B1" ? 50 : 44;
   674→   674→    const sceneCues = buildSingleLineCues(
   675→   675→      scene.narration,
   676→   676→      shotStart + (i === 0 ? lead : 0.15),
   677→   677→      shotEnd - 0.12,
   678→   678→      maxChars
   679→   679→    );
   680→   680→    captions.push(...sceneCues);
   681→   681→  });
   682→   682→
   683→   683→  const project: CanvasProject = {
   684→   684→    version: 1,
   685→   685→    style: input.styleB,
   686→   686→    width: plane.width,
   687→   687→    height: plane.height,
   688→   688→    fps: CANVAS_FPS,
   689→   689→    grid: input.styleB === "B1",
   690→   690→    voice: { path: "voice.wav", lead, tail, contains_sfx: false },
   691→   691→    shots,
   692→   692→    captions,
   693→   693→    sfx,
   694→   694→  };
   695→   695→
   696→   696→  const dir = path.join(jobDir(input.jobId), "canvas");
   697→   697→  fs.mkdirSync(dir, { recursive: true });
   698→   698→  const projectPath = path.join(dir, "project.json");
   699→   699→  fs.writeFileSync(projectPath, JSON.stringify(project));
   700→   700→  return { projectPath, duration, cues: captions };
   701→   701→}

   705→   705→export type RenderProgressLine = { shot?: number; total?: number };
   706→   706→
   707→   707→/**
   708→   708→ * Spawn the skill's canvas renderer for a project file. onShot reports
   709→   709→ * per-shot progress lines from the renderer.
   710→   710→ */
   711→   711→export function runCanvasRender(
   712→   712→  projectPath: string,
   713→   713→  onShot?: (line: RenderProgressLine) => void
   714→   714→): Promise<{ output: string; duration: number; frames: number }> {
   715→   715→  const wrapper = path.join(process.cwd(), "scripts", "render-autopilot.mjs");
   716→   716→  return new Promise((resolve, reject) => {
   717→   717→    const child = spawn("node", [wrapper, projectPath], {
   718→   718→      stdio: ["ignore", "pipe", "pipe"],
   719→   719→      env: { ...process.env, FORCE_COLOR: "0" },
   720→   720→    });
   721→   721→    let allOut = "";
   722→   722→    let lineBuf = "";
   723→   723→    let stderr = "";
   724→   724→    child.stdout.on("data", (b: Buffer) => {
   725→   725→      const text = b.toString();
   726→   726→      allOut += text;
   727→   727→      lineBuf += text;
   728→   728→      const lines = lineBuf.split("\n");
   729→   729→      lineBuf = lines.pop() ?? "";
   730→   730→      for (const line of lines) {
   731→   731→        const trimmed = line.trim();
   732→   732→        if (!trimmed.startsWith("{")) continue;
   733→   733→        try {
   734→   734→          const parsed = JSON.parse(trimmed) as RenderProgressLine;
   735→   735→          if (typeof parsed.shot === "number" && onShot) onShot(parsed);
   736→   736→        } catch {
   737→   737→          // bukan baris progress — abaikan
   738→   738→        }
   739→   739→      }
   740→   740→    });
   741→   741→    child.stderr.on("data", (b: Buffer) => (stderr = (stderr + b.toString()).slice(-3000)));
   742→   742→    child.on("error", reject);
   743→   743→    child.on("close", (code) => {
   744→   744→      if (code !== 0) {
   745→   745→        reject(new Error(`Canvas renderer exit ${code}: ${stderr || "tanpa stderr"}`));
   746→   746→        return;
   747→   747→      }
   748→   748→      const marker = allOut.indexOf("REPORT_JSON:");
   749→   749→      if (marker === -1) {
   750→   750→        reject(new Error(`Laporan renderer tidak ditemukan: ${stderr || allOut.slice(-400)}`));
   751→   751→        return;
   752→   752→      }
   753→   753→      try {
   754→   754→        const report = JSON.parse(allOut.slice(marker + "REPORT_JSON:".length).trim());
   755→   755→        resolve(report);
   756→   756→      } catch (err) {
   757→   757→        reject(new Error(`Laporan renderer tidak valid: ${err instanceof Error ? err.message : err}`));
   758→   758→      }
   759→   759→    });
   760→   760→  });
   761→   761→}
